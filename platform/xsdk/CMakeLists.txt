cmake_minimum_required(VERSION 3.25)
project(xds360_xsdk LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(XDS_HOST_TESTS "Build only the NooDS core tests on a PC" OFF)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT XSDK AND NOT XDS_HOST_TESTS)
  message(FATAL_ERROR "Use xsdk's cmake/xsdk.cmake toolchain file; see platform/xsdk/LEEME.md")
endif()
# This committed source set is the same core as the working xdk-port branch.
file(READ "${REPO}/platform/xenon/noods_sources.mk" source_list)
string(REGEX MATCHALL "[a-z_0-9/]+\\.cpp" source_names "${source_list}")
set(core_sources)
foreach(source IN LISTS source_names)
  if(XSDK AND source STREQUAL "gpu/gpu_3d_renderer.cpp")
    # Keep the XDK/upstream files untouched. Place real libstdc++ workers on
    # distinct physical cores from inside their entry, before any rendering.
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${REPO}/xdk/noods/core/${source}")
    file(READ "${REPO}/xdk/noods/core/${source}" renderer)
    string(REPLACE "void Gpu3DRenderer::drawThreaded(int thread) {"
      "void Gpu3DRenderer::drawThreaded(int thread) {\n    (void)xsdk_thread_move(thread == 0 ? 2 : 4);" renderer "${renderer}")
    string(REPLACE "../core.h" "core.h" renderer "${renderer}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gpu_3d_renderer.cpp"
      "#include <xsdk/thread.h>\n${renderer}")
    list(APPEND core_sources "${CMAKE_CURRENT_BINARY_DIR}/gpu_3d_renderer.cpp")
  else()
    list(APPEND core_sources "${REPO}/xdk/noods/core/${source}")
  endif()
endforeach()
add_library(noods_xsdk STATIC ${core_sources})
target_include_directories(noods_xsdk PUBLIC "${CMAKE_CURRENT_LIST_DIR}/compat")
target_include_directories(noods_xsdk SYSTEM PUBLIC "${REPO}/xdk/noods/core")
target_compile_definitions(noods_xsdk PUBLIC NOODS_XENON=1 LOG_LEVEL=0)
# The converted aligned byte-array accessors inherited MSVC's aliasing model.
target_compile_options(noods_xsdk PRIVATE -O2 -fno-strict-aliasing -fno-rtti)
if(XSDK)
  target_compile_definitions(noods_xsdk PUBLIC ENDIAN_BIG=1)
  add_executable(xds360-xsdk main.cpp "${REPO}/src/controller_mapper.cpp")
  target_include_directories(xds360-xsdk PRIVATE "${REPO}/include")
  target_link_libraries(xds360-xsdk PRIVATE noods_xsdk)
  target_compile_options(xds360-xsdk PRIVATE -O2 -Wall -Wextra)
  xsdk_add_shaders(xds360-xsdk shaders/screen.vert shaders/screen.frag)
  xsdk_add_xex(xds360-xsdk NAME "xDS360 xSDK" VERSION 0.1.0.0)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(noods_xsdk PUBLIC Threads::Threads)
  add_executable(xds-xsdk-core-test "${REPO}/tests/noods_core_smoke.cpp")
  target_link_libraries(xds-xsdk-core-test PRIVATE noods_xsdk)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(rom "${CMAKE_CURRENT_BINARY_DIR}/test.nds")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${REPO}/tests/make_noods_smoke_rom.py" "${rom}" COMMAND_ERROR_IS_FATAL ANY)
  enable_testing()
  add_test(NAME core COMMAND xds-xsdk-core-test "${rom}")
  add_test(NAME framepath COMMAND xds-xsdk-core-test "${rom}" --framepath-regression)
  add_test(NAME capture COMMAND xds-xsdk-core-test "${rom}" --capture-regression)
  add_test(NAME vram COMMAND xds-xsdk-core-test "${rom}" --capture-bank-matrix)
  add_test(NAME cadence COMMAND xds-xsdk-core-test "${rom}" --cadence-stress 2000)
endif()
